cmake_minimum_required(VERSION 3.20)
project(srw64_runtime_foundation VERSION 0.2.0 LANGUAGES C CXX)
# The default target is deliberately ROM/RT64/Python-free. It is NOT the game.
include(CTest)
add_subdirectory(src/native/app)
if(BUILD_TESTING)
    add_executable(srw64-app-test tests/native_app.cpp)
    target_link_libraries(srw64-app-test PRIVATE srw64_app)
    if(MSVC)
        target_compile_options(srw64-app-test PRIVATE /W4 /utf-8)
    else()
        target_compile_options(srw64-app-test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME native-app COMMAND srw64-app-test)
endif()

set(SRW64_APP_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing the pinned RT64 json/json.hpp for ROM-free bootstrap tests")
if(SRW64_APP_JSON_INCLUDE_DIR)
    srw64_add_launch("${SRW64_APP_JSON_INCLUDE_DIR}")
    if(BUILD_TESTING)
        add_executable(srw64-launch-test tests/native_launch.cpp)
        target_include_directories(srw64-launch-test PRIVATE "${SRW64_APP_JSON_INCLUDE_DIR}")
        target_link_libraries(srw64-launch-test PRIVATE srw64_launch)
        if(MSVC)
            target_compile_options(srw64-launch-test PRIVATE /W4 /utf-8)
        endif()
        add_test(NAME native-launch COMMAND srw64-launch-test)
    endif()
else()
    message(STATUS "Content/bootstrap tests need SRW64_APP_JSON_INCLUDE_DIR; session/path/hash tests remain enabled")
endif()
