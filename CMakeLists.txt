cmake_minimum_required(VERSION 3.20)
project(srw64_runtime_foundation VERSION 0.2.0 LANGUAGES C CXX)
# Base tests are ROM/RT64/Python-free. This is NOT the game build entry.
include(CTest)
add_subdirectory(src/native/app)
if(BUILD_TESTING)
    add_executable(srw64-app-test tests/native_app.cpp)
    target_link_libraries(srw64-app-test PRIVATE srw64_app)
    add_executable(srw64-import-codec-test tests/native_rom_import_codec.cpp)
    target_link_libraries(srw64-import-codec-test PRIVATE srw64_app)
    foreach(target srw64-app-test srw64-import-codec-test)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
    add_test(NAME native-app COMMAND srw64-app-test)
    add_test(NAME native-import-codec COMMAND srw64-import-codec-test)
endif()
set(SRW64_APP_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing the pinned RT64 json/json.hpp")
if(SRW64_APP_JSON_INCLUDE_DIR)
    srw64_add_launch("${SRW64_APP_JSON_INCLUDE_DIR}")
    if(BUILD_TESTING)
        find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
        add_executable(srw64-launch-test tests/native_launch.cpp)
        add_executable(srw64-import-probe tests/native_rom_import_probe.cpp)
        foreach(target srw64-launch-test srw64-import-probe)
            target_include_directories(${target} PRIVATE "${SRW64_APP_JSON_INCLUDE_DIR}")
            target_link_libraries(${target} PRIVATE srw64_launch)
            if(MSVC)
                target_compile_options(${target} PRIVATE /W4 /utf-8)
            endif()
        endforeach()
        add_test(NAME native-launch COMMAND srw64-launch-test)
        add_test(NAME native-import-oracle COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_import.py" --probe "$<TARGET_FILE:srw64-import-probe>")
    endif()
else()
    message(STATUS "Bootstrap/import tests require SRW64_APP_JSON_INCLUDE_DIR; base/codec tests remain enabled")
endif()
