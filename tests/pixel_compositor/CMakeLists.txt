cmake_minimum_required(VERSION 3.20)
project(srw64_pixel_gpu_test LANGUAGES C CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SRW64_RT64_HEADERS "" CACHE PATH "Pinned RT64 source with Plume/DXC submodules")
if(NOT EXISTS "${SRW64_RT64_HEADERS}/src/contrib/plume/plume_render_interface.h")
    message(FATAL_ERROR "SRW64_RT64_HEADERS must contain the pinned Plume source")
endif()
set(PLUME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PLUME_SDL_VULKAN_ENABLED OFF CACHE BOOL "" FORCE)
add_subdirectory("${SRW64_RT64_HEADERS}/src/contrib/plume" plume)
add_subdirectory("${SRW64_RT64_HEADERS}/src/tools/file_to_c" file-to-c)
if(APPLE)
    # Same pinned converter as the game, without its tool CMake's global output overrides.
    set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SRW64_RT64_HEADERS}/src/contrib/spirv-cross" spirv-cross)
    add_executable(spirv_cross_msl "${SRW64_RT64_HEADERS}/src/tools/spirv_cross_msl/spirv_cross_msl.cpp")
    target_link_libraries(spirv_cross_msl PRIVATE spirv-cross-core spirv-cross-msl)
    target_include_directories(spirv_cross_msl PRIVATE "${SRW64_RT64_HEADERS}/src/contrib/spirv-cross")
    # Test the upstream binary shader path; the game's existing source-shader
    # patch remains separate and is not applied to this pristine dependency.
    set(SRW64_METAL_SOURCE_SHADERS OFF)
endif()
include("${ROOT}/cmake/PixelCompositor.cmake")
srw64_add_pixel_compositor("${SRW64_RT64_HEADERS}")
add_executable(srw64-pixel-gpu-test gpu.cpp)
target_link_libraries(srw64-pixel-gpu-test PRIVATE srw64_pixel_compositor)
target_compile_features(srw64-pixel-gpu-test PRIVATE cxx_std_20)
if(APPLE)
    target_include_directories(srw64-pixel-gpu-test PRIVATE "${SRW64_RT64_HEADERS}/src/contrib/plume/contrib/metal-cpp")
    target_compile_options(srw64-pixel-gpu-test PRIVATE -fblocks)
    target_link_libraries(srw64-pixel-gpu-test PRIVATE "-framework Metal" "-framework Foundation" "-framework QuartzCore" "-framework Cocoa")
elseif(WIN32)
    target_compile_definitions(srw64-pixel-gpu-test PRIVATE NOMINMAX)
    target_link_libraries(srw64-pixel-gpu-test PRIVATE d3d12 dxgi dxguid)
else()
    find_package(X11 REQUIRED)
    target_link_libraries(srw64-pixel-gpu-test PRIVATE ${CMAKE_DL_LIBS} ${X11_LIBRARIES} ${X11_Xrandr_LIB})
endif()
add_test(NAME pixel-gpu COMMAND srw64-pixel-gpu-test)
set_tests_properties(pixel-gpu PROPERTIES TIMEOUT 120)
